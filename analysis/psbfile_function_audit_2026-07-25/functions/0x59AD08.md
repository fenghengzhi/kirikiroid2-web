deletingDestructor(self):                              // self 必须有效；无 null guard
  self.vptr = ncbInstanceAdaptor<PSBFile>::vtable
  if self._instance != null && self._sticky == false:
    owner = self._instance->owner

      TJSAlignedDealloc(owner.rawData); operator delete(owner)
