completeDestructor(self):                              // self 必须有效；无 null guard
  native = self._instance; self.vptr = typedAdaptorVtable
  if native != null && self._sticky == false:
    owner = native.owner
    if owner != null: --owner.refCount; if owner.refCount == 0: aligned_dealloc(owner.data); delete owner
    delete native
  self.vptr = tTJSNativeInstanceVtable; self._instance = null; self._sticky = false
  return                                                // complete 变体不 delete self；无 status/default/fallback
  if nested cleanup throws: catch; terminate            // @59AD04 -> @520FAC
