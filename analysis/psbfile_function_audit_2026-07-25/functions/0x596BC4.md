strings = header->strings; stringsData = header->stringsData
widthTag = *(strings + strings[0] - 11); values = strings + strings[0] - 10
switch node[0]:                                                        // node 无 null guard
  0x15 -> index = node[1]; 0x16 -> index = load_u16_unaligned(node+1)
