    if membername != null: return TJS_E_MEMBERNOTFOUND
    getter = self.getter; if getter == null: return TJS_E_ACCESSDENYED
    if objthis == null: return TJS_E_NATIVECLASSCRASH
    if result != null: result.Clear()
    adaptor = null; hr = objthis.NativeInstanceSupport(GETINSTANCE, PSBFileClassID, &adaptor)
    if FAILED(hr) or adaptor == null or adaptor.instance == null: return TJS_E_NATIVECLASSCRASH
    ok = rootTypedInvoke(result, 0, null, getter, adaptor.instance)  // direct @59B48C
